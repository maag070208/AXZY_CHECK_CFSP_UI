1. ITThemeProvider
================================================================================
Inyecta dinámicamente variables CSS y clases de Tailwind CSS para soporte de Modo Claro y Oscuro.

### Props (ITThemeProviderProps)
```typescript
export interface ITThemeConfig {
  primaryColor?: string;
  backgroundColor?: string;
  textColor?: string;
  sidebar?: {
    backgroundColor?: string;
    borderColor?: string;
    label?: { color?: string; size?: string; weight?: string };
    icon?: { color?: string; size?: string };
    active?: { backgroundColor?: string; color?: string; iconColor?: string };
    hover?: { backgroundColor?: string };
    badge?: { backgroundColor?: string; color?: string };
  };
  layout?: {
    backgroundColor?: string;
    contentPadding?: string;
  };
}

export interface ITThemeProviderProps {
  theme?: Partial<ITThemeConfig>;
  children: React.ReactNode;
}
```

### Ejemplo Real
```tsx
import React from 'react';
import { ITThemeProvider } from 'axzy-ui-system';

const customTheme = {
  primaryColor: '#6366f1', // Indigo-500
  sidebar: {
    backgroundColor: '#0f172a', // Slate-900 para sidebar oscuro siempre
    borderColor: '#1e293b',
    active: {
      backgroundColor: '#1e1b4b',
      iconColor: '#818cf8',
      color: '#e0e7ff'
    }
  }
};

export default function AppProvider({ children }) {
  return (
    <ITThemeProvider theme={customTheme}>
      {children}
    </ITThemeProvider>
  );
}
```

### Props (ITInputProps)
```typescript
export interface ITInputProps {
  name: string;
  type?: 'text' | 'password' | 'number' | 'email' | 'textarea' | 'checkbox';

  placeholder?: string;
  value?: any;
  onChange: (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => void;
  onBlur?: (e: React.FocusEvent<HTMLInputElement | HTMLTextAreaElement>) => void;
  variant?: 'primary' | 'secondary' | 'success' | 'danger' | 'warning' | 'info';
  iconLeft?: React.ReactNode;
  error?: string | boolean;
  required?: boolean;
  disabled?: boolean;
  className?: string;
}
```

### Ejemplo Real
```tsx
import React, { useState } from 'react';
import { ITInput } from 'axzy-ui-system';
import { FaEnvelope } from 'react-icons/fa';

export default function EmailInput() {
  const [email, setEmail] = useState('');
  const [error, setError] = useState<string | boolean>('');

  const validate = () => {
    if (!email) {
      setError('Este campo es requerido');
    } else if (!/\S+@\S+\.\S+/.test(email)) {
      setError('Correo electrónico inválido');
    } else {
      setError('');
    }
  };

  return (
    <ITInput
      name="email"
      type="email"
      label="Correo Electrónico"
      placeholder="ejemplo@axzy.dev"
      value={email}
      iconLeft={<FaEnvelope />}
      onChange={(e) => {
        setEmail(e.target.value);
        if (error) setError(''); // Limpia error al escribir
      }}
      onBlur={validate}
      error={error}
      required={true}
    />
  );
}
```

import React, { useState } from 'react';
import { ITSearchSelect, ITSearchSelectOption } from 'axzy-ui-system';

export default function AsyncUserSelector() {
  const [selectedVal, setSelectedVal] = useState<string | number>('');
  const [loading, setLoading] = useState(false);
  const [options, setOptions] = useState<ITSearchSelectOption[]>([
    { label: 'Asael Amaro', value: 1 },
    { label: 'Alex Dev', value: 2 },
    { label: 'Sofía Ramos', value: 3 }
  ]);

  const handleSearch = async (query: string) => {
    if (!query) return;
    setLoading(true);
    // Simulación de Fetch a API externa
    setTimeout(() => {
      setOptions([
        { label: `Resultado ${query} 1`, value: 101 },
        { label: `Resultado ${query} 2`, value: 102 }
      ]);
      setLoading(false);
    }, 800);
  };

  return (
    <ITSearchSelect
      label="Buscar Colaborador"
      placeholder="Escribe nombre para filtrar..."
      options={options}
      value={selectedVal}
      isLoading={loading}
      onSearch={handleSearch}
      onChange={(val) => setSelectedVal(val)}
      required={true}
    />
  );
}

import React, { useState } from 'react';
import { ITCalendar, CalendarEvent } from 'axzy-ui-system';

export default function MeetingScheduler() {
  const [events, setEvents] = useState<CalendarEvent[]>([
    {
      id: '1',
      title: 'Daily Standup',
      start: new Date(new Date().setHours(9, 0)),
      end: new Date(new Date().setHours(9, 30)),

    const title = prompt('Título del nuevo evento:');
    if (title) {
      setEvents(prev => [
        ...prev,
        {
          id: Date.now().toString(),
          title,
          start: date,
          end: new Date(date.getTime() + 60 * 60 * 1000), // 1 hora de duración
          color: '#8b5cf6'
        }
      ]);
    }
  };

  return (
    <div className="h-[600px] border border-slate-200 dark:border-slate-800 rounded-2xl p-4 bg-white dark:bg-slate-900">
      <ITCalendar
        events={events}
        mode="week"
        onSlotClick={handleSlotClick}
        onEventClick={(evt) => alert(`Evento: ${evt.title}`)}
        variant="purple"
      />
    </div>
  );
}

export enum UploadStatus {
  IDLE = 'idle',
  UPLOADING = 'uploading',
  SUCCESS = 'success',

export interface ITDropfileProps {
  onFileSelect: (file: File) => void;
  status?: 'idle' | 'uploading' | 'success' | 'error' | UploadStatus;

  errorMessage?: string;

### Ejemplo Real
```tsx
import React, { useState } from 'react';
import { ITDropfile, UploadStatus } from 'axzy-ui-system';

export default function DocumentUploader() {
  const [status, setStatus] = useState<UploadStatus>(UploadStatus.IDLE);
  const [progress, setProgress] = useState(0);

  const handleFile = (file: File) => {
    setStatus(UploadStatus.UPLOADING);
    setProgress(0);

    const interval = setInterval(() => {
      setProgress(p => {
        if (p >= 100) {
          clearInterval(interval);
          setStatus(UploadStatus.SUCCESS);
          return 100;
        }
        return p + 20;
      });
    }, 300);
  };

  return (
    <ITDropfile
      onFileSelect={handleFile}
      status={status}
      progress={progress}
      accept=".pdf,.png,.jpg"
      maxSizeMB={5}
    />
  );
}
```
